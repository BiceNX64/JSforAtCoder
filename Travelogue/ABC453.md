## 🔠 [ABC453-C](https://atcoder.jp/contests/abc453/tasks/abc453_c)

```javascript
// ABC453
// https://atcoder.jp/contests/abc453/tasks/abc453_c
// 2026-04-11

'use strict';
const fs = require('fs');

function Main(input) {
  const args = input.trim().split(/\s+/);
  let cur = 0;
  const next = () => args[cur++];
  const nextNum = () => Number(next());
  const nextBigInt = () => BigInt(next());

  // 👇️ ロジックを記述 👇️

  const N = nextNum();
  const L = [];
  for (let i = 0; i < N; i++) {
    L.push(nextNum());
  }

  let maxPasses = 0;

  // 再帰関数（深さ優先探索）の定義
  // index: 現在の移動回数, pos: 現在の座標, count: 現在の通過回数
  function dfs(index, pos, count) {
    // 全ての移動（N回）が終わったとき、最大値を更新して終了
    if (index === N) {
      if (count > maxPasses) {
        maxPasses = count;
      }
      return;
    }

    const dist = L[index];

    // パターン1: 正の方向（+）へ移動
    const nextPosPlus = pos + dist;
    const addedPlus = (pos * nextPosPlus < 0) ? 1 : 0;
    dfs(index + 1, nextPosPlus, count + addedPlus);

    // パターン2: 負の方向（-）へ移動
    const nextPosMinus = pos - dist;
    const addedMinus = (pos * nextPosMinus < 0) ? 1 : 0;
    dfs(index + 1, nextPosMinus, count + addedMinus);
  }

  // 初期条件：座標0.5からスタート、移動回数0、通過回数0
  dfs(0, 0.5, 0);

  console.log(maxPasses);

  // 👆️ ロジックを記述 👆️
}

Main(
  require('fs').readFileSync(
    process.platform === 'linux' ? '/dev/stdin' : process.argv[2],
    'utf8',
  ),
);
```

最初は常にボーダーとなる`0`を踏み越える方向に動くような手順を想定したのですが、数値によっては必ずしもその手順が踏み越える回数が最大となるとは限りません。Nの最大値が20であり、それぞれの数値を足す、または、引く、の2通りで2の20乗しかないので全部調べていくのだという方針までは考え、また以前やった順列を生成する再帰関数の考え方が応用できそうだというところまでは考えたのですが、力不足で実装できませんでした。

踏み越えたかどうかの判定の部分は、元のポジションと新しいポジションがそれぞれ異符号であれば積がマイナスになり、同符号であれば積がプラスになる、という判定でカウントをプラスしています。

````javascript
    const addedPlus = (pos * nextPosPlus < 0) ? 1 : 0;
````

別解としては、ビット全探索があります。わたしはビット全探索は初めてで理解できていませんが典型的なビット全探索の問題だったようで、単純な当て嵌めで解けるのであれば知っているか知らないかの差というのが大きいと思いました。

```javascript
const N = nextNum();
  const L = [];
  for (let i = 0; i < N; i++) {
    L.push(nextNum());
  }

  let maxPasses = 0;

  // 1をN回左にずらすことで、2のN乗（全パターン数）を計算
  const totalPatterns = 1 << N;

  // 0 から (2^N - 1) まで全てのパターンをループで試す
  for (let i = 0; i < totalPatterns; i++) {
    let currentPos = 0.5;
    let passCount = 0;

    for (let j = 0; j < N; j++) {
      // i の j 番目のビットが 1 か 0 かで、今回の移動方向を決める
      // (i >> j) & 1 が 1 なら正、0 なら負の方向へ
      const direction = (i >> j) & 1;
      const moveDist = L[j];
      
      let nextPos;
      if (direction === 1) {
        nextPos = currentPos + moveDist;
      } else {
        nextPos = currentPos - moveDist;
      }

      if (currentPos * nextPos < 0) {
        passCount++;
      }
      currentPos = nextPos;
    }

    // 全パターンのうち、最も通過回数が多かったものを保持
    if (passCount > maxPasses) {
      maxPasses = passCount;
    }
  }

  console.log(maxPasses);
```

JavaScriptのビット演算子は、内部的に数値を32ビット整数として扱うため、今回のN=20のような緩い制約であれば迷わずビット全探索をしたほうが楽で、複雑な条件の制御が必要な場合は再帰関数を利用したほうがいい場合があるようです。ビット全探索については別途ゼロから学びます。

Given that N is at most 20 and each number can either be added or subtracted, there are only 2^20 patterns. I figured out that an exhaustive search was necessary and thought I could apply the recursive logic for generating permutations that I learned before, but I lacked the implementation skills to pull it off.

To check if the border was crossed, I incremented the count if the product of the previous position and the new position was negative (indicating different signs).


An alternative solution is an exhaustive bit search (bit brute force). This was my first time encountering bit search, and although I don't fully understand it yet, it seems this was a typical problem for it. It made me realize the significance of whether you know a specific technique or not.

```javascript
const N = nextNum();
  const L = [];
  for (let i = 0; i < N; i++) {
    L.push(nextNum());
  }

  let maxPasses = 0;

  // Calculate 2^N (total patterns) by shifting 1 left N times
  const totalPatterns = 1 << N;

  // Loop through all patterns from 0 to (2^N - 1)
  for (let i = 0; i < totalPatterns; i++) {
    let currentPos = 0.5;
    let passCount = 0;

    for (let j = 0; j < N; j++) {
      // Determine the direction of movement based on whether the j-th bit of i is 1 or 0
      // (i >> j) & 1: 1 for positive, 0 for negative
      const direction = (i >> j) & 1;
      const moveDist = L[j];
      
      let nextPos;
      if (direction === 1) {
        nextPos = currentPos + moveDist;
      } else {
        nextPos = currentPos - moveDist;
      }

      if (currentPos * nextPos < 0) {
        passCount++;
      }
      currentPos = nextPos;
    }

    // Keep track of the maximum passCount among all patterns
    if (passCount > maxPasses) {
      maxPasses = passCount;
    }
  }

  console.log(maxPasses);
```

Since JavaScript bitwise operators treat numbers as 32-bit integers, an exhaustive bit search is much easier for loose constraints like N=20. Recursive functions might be better for cases requiring complex conditional control. I will learn bit search from scratch separately.
